CREATE DATABASE BibliotecaDigital;
USE BibliotecaDigital;

CREATE TABLE Categorias (
    id_categoria INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(100) NOT NULL
);

CREATE TABLE Autores (
    id_autor INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(150) NOT NULL
);

CREATE TABLE Livros (
    id_livro INT PRIMARY KEY AUTO_INCREMENT,
    titulo VARCHAR(255) NOT NULL,
    isbn VARCHAR(20),
    ano_publicacao INT
);

CREATE TABLE Livro_Autor (
    id_livro INT,
    id_autor INT,
    PRIMARY KEY (id_livro, id_autor),
    FOREIGN KEY (id_livro) REFERENCES Livros(id_livro),
    FOREIGN KEY (id_autor) REFERENCES Autores(id_autor)
);

CREATE TABLE Livro_Categoria (
    id_livro INT,
    id_categoria INT,
    PRIMARY KEY (id_livro, id_categoria),
    FOREIGN KEY (id_livro) REFERENCES Livros(id_livro),
    FOREIGN KEY (id_categoria) REFERENCES Categorias(id_categoria)
);

CREATE TABLE Usuarios (
    id_usuario INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(150) NOT NULL,
    email VARCHAR(100) UNIQUE,
    data_cadastro DATE
);

CREATE TABLE Funcionarios (
    id_funcionario INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(150) NOT NULL,
    cargo VARCHAR(50),
    data_admissao DATE
);

CREATE TABLE Exemplares (
    id_exemplar INT PRIMARY KEY AUTO_INCREMENT,
    id_livro INT,
    status ENUM('disponivel', 'emprestado', 'manutencao'),
    FOREIGN KEY (id_livro) REFERENCES Livros(id_livro)
);

CREATE TABLE Emprestimos (
    id_emprestimo INT PRIMARY KEY AUTO_INCREMENT,
    id_usuario INT,
    id_exemplar INT,
    id_funcionario INT,
    data_emprestimo DATETIME,
    data_devolucao_prevista DATE,
    FOREIGN KEY (id_usuario) REFERENCES Usuarios(id_usuario),
    FOREIGN KEY (id_exemplar) REFERENCES Exemplares(id_exemplar),
    FOREIGN KEY (id_funcionario) REFERENCES Funcionarios(id_funcionario)
);



CREATE TABLE Devolucoes (
    id_devolucao INT PRIMARY KEY AUTO_INCREMENT,
    id_emprestimo INT UNIQUE, -- garante 1 devolução por empréstimo
    data_devolucao_real DATETIME NOT NULL,
    observacoes VARCHAR(255),

    FOREIGN KEY (id_emprestimo) REFERENCES Emprestimos(id_emprestimo)
);


CREATE TABLE Pendencias (
    id_pendencia INT PRIMARY KEY AUTO_INCREMENT,
    id_emprestimo INT NOT NULL,
    
    tipo ENUM('atraso', 'multa', 'dano', 'perda') NOT NULL,
    descricao VARCHAR(255),
    
    valor_multa DECIMAL(10,2) DEFAULT 0.00,
    
    status ENUM('pendente', 'pago', 'resolvido') DEFAULT 'pendente',
    
    data_registro DATETIME DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (id_emprestimo) REFERENCES Emprestimos(id_emprestimo)
);

SELECT 

 -- LIVRO
l.id_livro,
l.titulo,
l.isbn,
l.ano_publicacao,

     -- AUTOR 
    a.nome AS autor,

     -- CATEGORIA 
    c.nome AS categoria,

    -- EXEMPLAR 
    ex.id_exemplar,
    ex.status AS status_exemplar,

   --  USUÁRIO 
    u.id_usuario,
    u.nome AS usuario,
    u.email,
    u.data_cadastro,

   --  FUNCIONÁRIO 
f.id_funcionario,
f.nome AS funcionario,
f.cargo,
f.data_admissao,

    -- EMPRÉSTIMO 
    e.id_emprestimo,

    DATE_FORMAT(
        e.data_emprestimo,
        '%d/%m/%Y %H:%i'
    ) AS data_emprestimo,

    DATE_FORMAT(
        e.data_devolucao_prevista,
        '%d/%m/%Y'
    ) AS devolucao_prevista,

-- DEVOLUÇÃO
d.id_devolucao,
d.data_devolucao_real,
d.observacoes,

    -- SITUAÇÃO DA DEVOLUÇÃO 
    CASE



        WHEN d.data_devolucao_real > e.data_devolucao_prevista
            THEN 'Devolvido com atraso'

        ELSE 'Devolvido no prazo'

    END AS situacao_devolucao,

    
   /* PENDÊNCIAS */
/* PENDÊNCIAS */
p.id_pendencia,
p.tipo AS tipo_pendencia,
p.descricao,
p.valor_multa,
p.status AS status_pendencia,

DATE_FORMAT(
    p.data_registro,
    '%d/%m/%Y %H:%i'
) AS data_registro,

   --  CLASSIFICAÇÃO DA MULTA 
    CASE

        WHEN p.valor_multa >= 20
            THEN 'Multa Alta'

        WHEN p.valor_multa >= 5
            THEN 'Multa Média'

        WHEN p.valor_multa > 0
            THEN 'Multa Baixa'

        ELSE 'Sem Multa'

    END AS classificacao_multa,

  --  TOTAL DE EMPRÉSTIMOS 
    (
        SELECT COUNT(*)
        FROM Emprestimos
    ) AS total_emprestimos,

 --    TOTAL DE PENDÊNCIAS 
    (
        SELECT COUNT(*)
        FROM Pendencias pe
        WHERE pe.status = 'pendente'
    ) AS total_pendencias_ativas

FROM Livros l

LEFT JOIN Livro_Autor la
    ON l.id_livro = la.id_livro

LEFT JOIN Autores a
    ON la.id_autor = a.id_autor

LEFT JOIN Livro_Categoria lc
    ON l.id_livro = lc.id_livro

LEFT JOIN Categorias c
    ON lc.id_categoria = c.id_categoria

LEFT JOIN Exemplares ex
    ON l.id_livro = ex.id_livro

LEFT JOIN Emprestimos e
    ON ex.id_exemplar = e.id_exemplar

LEFT JOIN Usuarios u
    ON e.id_usuario = u.id_usuario

LEFT JOIN Funcionarios f
    ON e.id_funcionario = f.id_funcionario

LEFT JOIN Devolucoes d
    ON e.id_emprestimo = d.id_emprestimo

LEFT JOIN Pendencias p
    ON e.id_emprestimo = p.id_emprestimo

WHERE
    e.id_emprestimo IS NOT NULL

ORDER BY
    p.valor_multa DESC,
    e.data_emprestimo DESC,
    l.titulo ASC;
